cmake_minimum_required(VERSION 3.16)
project(BuoyGitKit C)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_CLI OFF CACHE BOOL "" FORCE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(USE_HTTPS OFF CACHE STRING "" FORCE)
set(USE_SSH OFF CACHE STRING "" FORCE)
set(USE_GSSAPI OFF CACHE BOOL "" FORCE)
set(USE_NTLMCLIENT OFF CACHE BOOL "" FORCE)
set(USE_ICONV OFF CACHE BOOL "" FORCE)
set(USE_SHA1 CollisionDetection CACHE STRING "" FORCE)
set(USE_SHA256 Builtin CACHE STRING "" FORCE)
set(USE_BUNDLED_ZLIB ON CACHE STRING "" FORCE)
set(REGEX_BACKEND builtin CACHE STRING "" FORCE)
add_subdirectory("${LIBGIT2_SOURCE}" libgit2-build EXCLUDE_FROM_ALL)
add_library(buoy_gitkit SHARED bridge.c)
set_target_properties(buoy_gitkit PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES)
target_include_directories(buoy_gitkit PRIVATE "${LIBGIT2_SOURCE}/include")
target_link_libraries(buoy_gitkit PRIVATE libgit2package)
if(APPLE)
    set_target_properties(buoy_gitkit PROPERTIES INSTALL_NAME_DIR "@rpath" BUILD_WITH_INSTALL_NAME_DIR YES)
endif()

if(WIN32)
    set_target_properties(buoy_gitkit PROPERTIES PREFIX "")
endif()
if(MINGW)
    target_link_options(buoy_gitkit PRIVATE -static-libgcc -static)
endif()
if(MSVC)
    set_target_properties(buoy_gitkit PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
